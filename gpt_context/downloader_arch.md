import os


from config import (
    IMAP_HOST,
    IMAP_PORT,
    IMAP_USERNAME,
    IMAP_PASSWORD,
    IMAP_MAILBOX,
    IMAP_TRASH,
    IMAP_SUBJECT,
    IMAP_OUTPUT_DIR,
    IMAP_TIMEOUT,
)


def download_news_email():
    """
    Скачивает одно письмо из INBOX, в заголовке которого
    присутствует IMAP_SUBJECT.

    Письмо сохраняется как оригинальный RFC822 .eml.

    Возвращает путь к сохранённому .eml.

    Если подходящих писем нет:
        возвращает None.

    При любой ошибке:
        выбрасывает исключение.
    """

    os.makedirs(IMAP_OUTPUT_DIR, exist_ok=True)

    mail = None
    temp_path = None
    output_path = None

    try:
        mail = imaplib.IMAP4_SSL(
            IMAP_HOST,
            IMAP_PORT,
            timeout=IMAP_TIMEOUT,
        )

        mail.login(
            IMAP_USERNAME,
            IMAP_PASSWORD,
        )

        status, _ = mail.select(IMAP_MAILBOX)

        if status != "OK":
            raise RuntimeError(
                f"Не удалось открыть почтовый ящик: {IMAP_MAILBOX}"
            )

        # IMAP SUBJECT ищет вхождение строки.
        # Поэтому для "ЗГП" будут найдены темы вроде:
        # "ЗГП"
        # "ЗГП №123"
        # "Новости ЗГП"
        status, data = mail.search(
            None,
            "SUBJECT",
            IMAP_SUBJECT,
        )

        if status != "OK":
            raise RuntimeError(
                "Ошибка поиска писем в IMAP."
            )

        message_ids = data[0].split()

        if not message_ids:
            return None

        # Берём самое старое подходящее письмо.
        # Это позволяет обрабатывать новости последовательно.
        message_id = message_ids[0]

        # Получаем UID письма для уникального имени файла.
        status, uid_data = mail.fetch(
            message_id,
            "(UID)",
        )

        if status != "OK":
            raise RuntimeError(
                "Не удалось получить UID письма."
            )

        uid = None

        for response_part in uid_data:
            if isinstance(response_part, tuple):
                response = response_part[0]

                if isinstance(response, bytes):
                    response = response.decode(
                        "ascii",
                        errors="ignore",
                    )

                parts = response.split()

                if "UID" in parts:
                    uid_index = parts.index("UID")

                    if uid_index + 1 < len(parts):
                        uid = parts[uid_index + 1]
                        break

        if not uid:
            raise RuntimeError(
                "Не удалось определить UID письма."
            )

        # Уникальное имя файла.
        filename = f"ЗГП_{uid}.eml"

        output_path = os.path.join(
            IMAP_OUTPUT_DIR,
            filename,
        )

        # Временный файл.
        # Он никогда не считается готовым .eml.
        temp_path = output_path + ".part"

        # Получаем оригинальный RFC822.
        status, data = mail.fetch(
            message_id,
            "(RFC822)",
        )

        if status != "OK":
            raise RuntimeError(
                "Не удалось получить содержимое письма."
            )

        raw_email = None

        for response_part in data:
            if isinstance(response_part, tuple):
                raw_email = response_part[1]
                break

        if raw_email is None:
            raise RuntimeError(
                "IMAP не вернул содержимое письма."
            )

        # Сначала пишем .part.
        with open(temp_path, "wb") as f:
            f.write(raw_email)

        # Только после полного успешного сохранения
        # превращаем .part в настоящий .eml.
        os.replace(
            temp_path,
            output_path,
        )

        temp_path = None

        print(
            f"Письмо скачано: {output_path}"
        )

        # Перемещаем письмо в корзину.
        status, _ = mail.copy(
            message_id,
            IMAP_TRASH,
        )

        if status != "OK":
            raise RuntimeError(
                f"Не удалось переместить письмо "
                f"в корзину: {IMAP_TRASH}"
            )

        # Помечаем оригинал в INBOX на удаление.
        status, _ = mail.store(
            message_id,
            "+FLAGS",
            "\\Deleted",
        )

        if status != "OK":
            raise RuntimeError(
                "Не удалось пометить исходное письмо "
                "на удаление."
            )

        # Удаляем оригинал из INBOX.
        mail.expunge()

        print(
            f"Письмо перемещено в корзину: {IMAP_TRASH}"
        )

        return output_path

    except Exception:
        # Если скачивание оборвалось,
        # .part обязательно удаляем.
        if temp_path and os.path.exists(temp_path):
            try:
                os.remove(temp_path)
            except OSError:
                pass

        raise

    finally:
        if mail is not None:
            try:
                mail.close()
            except Exception:
                pass

            try:
                mail.logout()
            except Exception:
                pass

Почему здесь .part

Это важная деталь.

Нельзя делать просто:

open("downloaded_eml/ЗГП.eml", "wb")

и считать, что наличие файла означает успешную загрузку.

При обрыве соединения можно получить вполне настоящий на вид ЗГП.eml, но только с частью сообщения.

Поэтому схема:

ЗГП_12345.eml.part
        ↓
полностью записали
        ↓
os.replace()
        ↓
ЗГП_12345.eml

Если IMAP оборвался посередине:

ЗГП_12345.eml.part
        ↓
удаляется

А полноценного .eml вообще не появляется.

6. Очистка downloaded_eml и extracted_images

Вот это уже действительно лучше сделать в main.py.

Нам нужна функция примерно такого вида:

import os
import glob


def cleanup_working_files():
    """
    Удаляет временные .eml и извлечённые изображения.
    Вызывается в finally.
    """

    # Удаляем скачанные письма
    for path in glob.glob(
        r".\downloaded_eml\*.eml"
    ):
        try:
            os.remove(path)
        except OSError as e:
            print(
                f"Не удалось удалить {path}: {e}"
            )

    # Удаляем временные файлы загрузки
    for path in glob.glob(
        r".\downloaded_eml\*.part"
    ):
        try:
            os.remove(path)
        except OSError as e:
            print(
                f"Не удалось удалить {path}: {e}"
            )

    # Удаляем извлечённые изображения
    for path in glob.glob(
        r".\extracted_images\*"
    ):
        try:
            if os.path.isfile(path):
                os.remove(path)
        except OSError as e:
            print(
                f"Не удалось удалить {path}: {e}"
            )

Но тут есть маленькая архитектурная тонкость.

Не надо делать finally вокруг всего цикла из 5 писем, если мы хотим очищать каждое письмо сразу после его обработки.

Лучше:

while True:

    скачать письмо

    если писем нет:
        break

    try:
        обработать письмо
    finally:
        удалить .eml
        удалить extracted_images

Тогда при падении обработки новости №3:

№1 → обработана → очищена
№2 → обработана → очищена
№3 → ошибка
№3 → временные файлы очищены
STOP
№4 → НЕ обрабатывается
№5 → НЕ обрабатывается

И это именно соответствует твоему требованию «при ошибке не было цикла».
7. Важный нюанс с extracted_images

У нас eml_parser.py сохраняет изображения именно в:

output_dir="extracted_images"

Поэтому на каждой итерации нужно после окончания работы новости очищать:

Это даже желательно делать в finally, потому что WordPress может упасть после того, как картинки уже были извлечены.

Иначе следующая новость может получить в галерею картинки предыдущей.

raise

на уровень главного цикла.

Фактически process_vk() должен работать примерно по принципу:

try:
    ...
    получить VK video
    ...
except Exception as e:
    print(f"Ошибка VK video: {e}")
    return

А main.py воспринимает это как:

    «VK-видео не получилось, ну ладно, остальная новость всё равно может быть опубликована».

Это соответствует описанной тобой текущей логике проекта. Сам vk_video.py отвечает именно за получение player, preview и данных видео, а main.py затем добавляет iframe и preview к новости.

Тогда результат:

№3 → опубликована
№4 → опубликована без видео
№5 → опубликована

Письмо №2 после этого:

INBOX
   ↓
прочитано
   ↓
перемещено в Trash

Ситуация №2: timeout при скачивании первого письма

Здесь я согласен с твоей логикой.

Например:

INBOX

[UNSEEN] ЗГП №1
[UNSEEN] ЗГП №2
[UNSEEN] обычное письмо
[UNSEEN] ЗГП №3

Скрипт берёт самое старое:

ЗГП №1

Пытается скачать:

attempt 1 → timeout
attempt 2 → timeout
attempt 3 → timeout

После третьей попытки:

ЗГП №1

не удаляется.

Но:

\Seen

ставится.

Получаем:

INBOX

[SEEN]   ЗГП №1       ← ошибка загрузки
[UNSEEN] ЗГП №2
[UNSEEN] обычное письмо
[UNSEEN] ЗГП №3

Затем цикл продолжает работу.

Следующим будет:

ЗГП №2

Если оно успешно обработалось:

ЗГП №2 → Trash

Потом:

ЗГП №3 → обработка → Trash

И в конце в INBOX останется:

[SEEN] ЗГП №1

То есть одним взглядом в почтовом ящике можно увидеть:

    Вот это письмо не обработалось.

Причём оно останется там именно потому, что произошло исключение.

Это довольно хороший механизм контроля без отдельной базы данных.
Но есть один важный нюанс

Я бы не помечал письмо прочитанным до успешного скачивания.

Для скачивания нужно использовать:

BODY.PEEK[]

или соответствующий IMAP-запрос, который не устанавливает \Seen.

И только после трёх неудачных попыток делать:

STORE +FLAGS (\Seen)

То есть:

попытка скачать
     ↓
timeout
     ↓
UNSEEN сохраняется

попытка 2
     ↓
timeout

попытка 3
     ↓
timeout
     ↓
ставим SEEN

Это особенно важно, потому что иначе сам FETCH (RFC822) потенциально может изменить статус сообщения, и наша схема контроля станет менее надёжной.
А что делать с ошибкой обработки всего письма?

Здесь я бы ввёл ещё одно правило.

Предположим, .eml скачался нормально:

ЗГП №1
ЗГП №2

Но create_post() для №2 получил, например, ошибку WordPress API.

Тогда:

№1 → опубликована
№2 → ошибка

Я не стал бы отправлять письмо в корзину.

Вместо этого:

письмо → оставить INBOX
пометить SEEN

Таким образом:

INBOX

[SEEN] ЗГП письмо №...

означает:

    «С этим письмом что-то пошло не так, посмотри журнал».

Это намного безопаснее, чем удалить письмо после частичной обработки.

Но тут появляется один вопрос: если на следующем запуске мы ищем только UNSEEN, проблемное письмо повторно обрабатываться не будет. И это, на мой взгляд, правильно для твоего сценария: иначе WordPress может получить дубль уже опубликованной новости.

То есть:

UNSEEN → работа
SEEN → больше автоматически не трогаем
TRASH → успешно завершено

Получается очень чёткая модель.
Тогда конечная машина состояний получается такой

             ┌──────────────────────────┐
             │                          │
             ▼                          │
        UNSEEN + ЗГП                    │
             │                          │
             │ скачать                  │
             │                          │
       ┌─────┴─────┐                    │
       │           │                    │
     успех       3× ошибка              │
       │           │                    │
       │           ▼                    │
       │         SEEN                    │
       │       + INBOX                   │
       │           │                     │
       │        STOP для                 │
       │        этого письма             │
       │                                 │
       ▼                                 │
  обработка новостей                     │
       │                                 │
       ├── отдельная ошибка новости      │
       │       ↓                         │
       │   продолжаем                    │
       │                                 │
       ├── всё успешно ──────────────► TRASH
       │                                 │
       │                                 │
       └── критическая ошибка ───────► SEEN + INBOX

При этом после SEEN + INBOX основной цикл может продолжить искать следующее UNSEEN письмо.
Что касается .pdf во вложении

Здесь как раз есть интересный момент.

Согласно текущему eml_parser.py, он рассматривает вложение следующим образом:

.docx → documents
.jpg/.jpeg/.png/.webp/.gif → images
всё остальное → игнорируется

В частности, список изображений сейчас ограничен этими расширениями, а .docx обрабатывается отдельно.

Поэтому если письмо содержит:

ЗГП.docx
photo1.jpg
photo2.jpg
document.pdf

получится:

documents
 └── ЗГП.docx

images
 ├── photo1.jpg
 └── photo2.jpg

PDF
 └── проигнорирован

То есть PDF сейчас не сохраняется вообще.

И это не вызовет ошибки.

То же самое будет, например, с:

.xlsx
.zip
.txt
.pptx
.mp4

если они идут именно как вложения.

При этом text/plain и text/html части самого письма продолжают обрабатываться, поскольку это не вложения в нашей текущей логике.
